cmake_minimum_required(VERSION 3.20)
project(free_camera_test C)
enable_testing()

# Exercise the actual manual override without linking the rest of the game.
set(manual "${CMAKE_CURRENT_SOURCE_DIR}/../../src/game/recomp/recomp_manual.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${manual}")
file(READ "${manual}" source)
string(FIND "${source}" "/* FREE_CAMERA_HOOK_BEGIN" begin)
string(FIND "${source}" "/* FREE_CAMERA_HOOK_END */" end)
if(begin LESS 0 OR end LESS begin)
    message(FATAL_ERROR "Free camera hook markers missing")
endif()
math(EXPR length "${end} - ${begin}")
string(SUBSTRING "${source}" ${begin} ${length} hook)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/camera_hook.inc" "${hook}")
add_executable(free_camera_test free_camera_test.c ../../src/game/free_camera.c)
target_include_directories(free_camera_test PRIVATE ../../src/game ../../src "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(free_camera_test PRIVATE user32)
add_test(NAME free_camera COMMAND free_camera_test)
