cmake_minimum_required(VERSION 3.20)
project(clipping_test C)
enable_testing()
set(renderer "${CMAKE_CURRENT_SOURCE_DIR}/../../src/nv2a/nv2a_pgraph_d3d11.c")
file(READ "${renderer}" source)
function(extract first last output)
  string(FIND "${source}" "${first}" begin)
  string(FIND "${source}" "${last}" end)
  if(begin LESS 0 OR end LESS begin)
    message(FATAL_ERROR "Renderer clipping extraction boundary changed")
  endif()
  math(EXPR length "${end}-${begin}")
  string(SUBSTRING "${source}" ${begin} ${length} part)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}" "${part}")
endfunction()
extract("/* XYZRHW/diffuse/UV0 prefix" "static struct {" vertices.h)
extract("typedef struct { float X, Y, Z, W;" "static int nv_clip_triangle_old" clip.h)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${renderer}")
add_executable(clipping_test clipping_test.c)
target_include_directories(clipping_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME clipping COMMAND clipping_test)
