project(perf_capture_test C)
enable_testing()
add_executable(perf_capture_test perf_capture_test.c ../../src/perf_capture.c ../../src/perf_frame_log.c)
target_include_directories(perf_capture_test PRIVATE ../../src)
add_test(NAME perf_capture COMMAND perf_capture_test)
set(HOST "${CMAKE_CURRENT_SOURCE_DIR}/../../src/game/main.c")
file(READ "${HOST}" SOURCE)
string(FIND "${SOURCE}" "static int s_perf_enabled;" BEGIN)
string(FIND "${SOURCE}" "void doa3_present_frame(void)" END)
if(BEGIN LESS 0 OR END LESS BEGIN)
    message(FATAL_ERROR "Frame capture extraction boundary changed")
endif()
math(EXPR LENGTH "${END}-${BEGIN}")
string(SUBSTRING "${SOURCE}" ${BEGIN} ${LENGTH} FRAMES)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/frame_impl.h" "${FRAMES}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${HOST}")
add_executable(perf_lifecycle_test perf_lifecycle_test.c ../../src/perf_capture.c ../../src/perf_frame_log.c)
