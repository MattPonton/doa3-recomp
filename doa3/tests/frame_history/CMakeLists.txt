cmake_minimum_required(VERSION 3.20)
project(frame_history_test C)
enable_testing()
add_executable(frame_history_test frame_history_test.c)
# Compile the real private frame-copy implementation without linking the
# entire game renderer. Regenerate whenever that implementation changes.
set(DEVICE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../src/d3d/d3d8_device.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${DEVICE_SOURCE}")
file(READ "${DEVICE_SOURCE}" DEVICE_CODE)
string(FIND "${DEVICE_CODE}" "typedef struct { ID3D11Texture2D *tex;" START)
string(FIND "${DEVICE_CODE}" "static HRESULT __stdcall dev_Clear(" END)
if(START LESS 0 OR END LESS START)
    message(FATAL_ERROR "Frame history implementation boundaries changed")
endif()
math(EXPR LENGTH "${END} - ${START}")
string(SUBSTRING "${DEVICE_CODE}" ${START} ${LENGTH} FRAME_CODE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/frame_history_impl.h" "${FRAME_CODE}")
target_include_directories(frame_history_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(frame_history_test PRIVATE d3d11 d3dcompiler)
add_test(NAME frame_history COMMAND frame_history_test)
