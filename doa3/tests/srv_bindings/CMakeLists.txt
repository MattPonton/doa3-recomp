cmake_minimum_required(VERSION 3.20)
project(srv_bindings_test C)
enable_testing()
set(DEVICE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../src/d3d/d3d8_device.c")
file(READ "${DEVICE_SOURCE}" SOURCE)
string(FIND "${SOURCE}" "static HRESULT __stdcall dev_SetTexture(" BEGIN)
string(FIND "${SOURCE}" "static HRESULT __stdcall dev_GetTexture(" END)
if(BEGIN LESS 0 OR END LESS BEGIN)
    message(FATAL_ERROR "Texture binding extraction boundary changed")
endif()
math(EXPR LENGTH "${END}-${BEGIN}")
string(SUBSTRING "${SOURCE}" ${BEGIN} ${LENGTH} BINDING)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/binding_impl.h" "${BINDING}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${DEVICE_SOURCE}")
add_executable(srv_bindings_test srv_bindings_test.c)
target_include_directories(srv_bindings_test PRIVATE ../../src/d3d "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(srv_bindings_test PRIVATE d3d11 dxguid)
add_test(NAME srv_bindings COMMAND srv_bindings_test)
