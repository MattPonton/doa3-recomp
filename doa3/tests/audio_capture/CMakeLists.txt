cmake_minimum_required(VERSION 3.20)
project(audio_capture_test C)
enable_testing()
set(SOURCE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../src/audio/oal_output.c")
file(READ "${SOURCE_PATH}" SOURCE)
string(FIND "${SOURCE}" "#define MIX_RING_FRAMES" BEGIN)
string(FIND "${SOURCE}" "/* Submit a buffer of mixed samples" END)
if(BEGIN LESS 0 OR END LESS BEGIN)
    message(FATAL_ERROR "Audio capture extraction boundary changed")
endif()
math(EXPR LENGTH "${END}-${BEGIN}")
string(SUBSTRING "${SOURCE}" ${BEGIN} ${LENGTH} CAPTURE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/capture_full.h" "${CAPTURE}")
string(REPLACE "#define MIX_RING_FRAMES (48000 * 30)" "#define MIX_RING_FRAMES 127" CAPTURE "${CAPTURE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/capture_impl.h" "${CAPTURE}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SOURCE_PATH}")
add_executable(audio_capture_test audio_capture_test.c)
target_include_directories(audio_capture_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME audio_capture COMMAND audio_capture_test)
add_executable(audio_capture_benchmark audio_capture_benchmark.c)
target_include_directories(audio_capture_benchmark PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
