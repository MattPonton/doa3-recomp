cmake_minimum_required(VERSION 3.20)
project(combiner_cache_test C)
enable_testing()
set(D3D "${CMAKE_CURRENT_SOURCE_DIR}/../../src/d3d")
set(SOURCE "${D3D}/d3d8_combiners.c")
file(READ "${SOURCE}" content)
function(extract first last output)
  string(FIND "${content}" "${first}" begin)
  string(FIND "${content}" "${last}" end)
  if(begin LESS 0 OR end LESS begin)
    message(FATAL_ERROR "Combiner extraction boundary changed")
  endif()
  math(EXPR length "${end}-${begin}")
  string(SUBSTRING "${content}" ${begin} ${length} part)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}" "${part}")
endfunction()
extract("#define COMBINER_CACHE_SIZE" "/* ================================================================\n * Color Helpers" cache_key.h)
extract("ID3D11PixelShader *d3d8_combiners_get_shader(" "/* ================================================================\n * Initialization / Shutdown" cache_lookup.h)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SOURCE}")
add_executable(combiner_cache_test combiner_cache_test.c)
target_include_directories(combiner_cache_test PRIVATE "${D3D}" "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME combiner_cache COMMAND combiner_cache_test)
