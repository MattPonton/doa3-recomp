cmake_minimum_required(VERSION 3.20)
project(reflection_mask_test C)
enable_testing()
function(extract file first last output)
 file(READ "${file}" content)
 string(FIND "${content}" "${first}" begin)
 string(FIND "${content}" "${last}" end)
 if(begin LESS 0 OR end LESS begin)
  message(FATAL_ERROR "Renderer extraction boundary changed: ${file}")
 endif()
 math(EXPR length "${end}-${begin}")
 string(SUBSTRING "${content}" ${begin} ${length} part)
 file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}" "${part}")
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${file}")
endfunction()
set(D3D "${CMAKE_CURRENT_SOURCE_DIR}/../../src/d3d")
set(PGRAPH "${CMAKE_CURRENT_SOURCE_DIR}/../../src/nv2a/nv2a_pgraph_d3d11.c")
extract("${D3D}/d3d8_shaders.c" "static const char g_vs_source[] =" "static const char g_ps_source[] =" vertex_shader.h)
extract("${D3D}/d3d8_combiners.c" "#define EMIT(fmt" "#undef EMIT" pixel_generator.h)
extract("${PGRAPH}" "static void nv_comb_input(" "/* Specular power" registers.h)
extract("${PGRAPH}" "/* XYZRHW/diffuse/UV0 prefix" "static struct {" vertices.h)
extract("${PGRAPH}" "typedef struct { float X, Y, Z, W;" "/* Clip one convex polygon" clip.h)
add_executable(reflection_mask_test reflection_mask_test.c)
target_include_directories(reflection_mask_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${D3D}")
target_link_libraries(reflection_mask_test PRIVATE d3d11 d3dcompiler)
add_test(NAME reflection_mask COMMAND reflection_mask_test)
