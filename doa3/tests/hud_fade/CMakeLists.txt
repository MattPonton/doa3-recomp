cmake_minimum_required(VERSION 3.20)
project(hud_fade_test C)
enable_testing()
set(NV "${CMAKE_CURRENT_SOURCE_DIR}/../../src/nv2a/nv2a_pgraph_d3d11.c")
file(READ "${NV}" content)
string(FIND "${content}" "static int nv_use_vertex_diffuse(" first)
string(FIND "${content}" "static void nv_apply_polygon_offset(" last)
if(first LESS 0 OR last LESS first)
  message(FATAL_ERROR "HUD diffuse extraction boundary changed")
endif()
math(EXPR length "${last}-${first}")
string(SUBSTRING "${content}" ${first} ${length} implementation)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/diffuse.h" "${implementation}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${NV}")
add_executable(hud_fade_test hud_fade_test.c)
target_include_directories(hud_fade_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME hud_fade COMMAND hud_fade_test)
