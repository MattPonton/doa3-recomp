cmake_minimum_required(VERSION 3.20)
project(texgen_test C)
enable_testing()
set(SHADER_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../src/d3d/d3d8_shaders.c" CACHE FILEPATH "Renderer shader source")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SHADER_SOURCE}")
file(READ "${SHADER_SOURCE}" SHADER_CODE)
string(FIND "${SHADER_CODE}" "static const char g_nv2aff_vs_source[] =" START)
if(START LESS 0)
    message(FATAL_ERROR "NV2A shader implementation boundary changed")
endif()
string(SUBSTRING "${SHADER_CODE}" ${START} -1 SHADER_IMPL)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/nv2aff_impl.h" "${SHADER_IMPL}")
add_executable(texgen_test texgen_test.c)
target_include_directories(texgen_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "../../src/d3d")
target_link_libraries(texgen_test PRIVATE d3d11 d3dcompiler)
add_test(NAME texgen COMMAND texgen_test)
