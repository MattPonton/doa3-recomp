cmake_minimum_required(VERSION 3.20)
project(apu_idle_test C)
enable_testing()
set(APU "${CMAKE_CURRENT_SOURCE_DIR}/../../src/apu")
set(SOURCE "${APU}/apu_vp.c")
file(READ "${SOURCE}" content)
string(FIND "${content}" "bool mcpx_apu_vp_poll_idle(" begin)
string(FIND "${content}" "void mcpx_apu_vp_frame(" end)
if(begin LESS 0 OR end LESS begin)
  message(FATAL_ERROR "APU idle walk extraction boundary changed")
endif()
math(EXPR length "${end}-${begin}")
string(SUBSTRING "${content}" ${begin} ${length} body)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/idle_walk.h" "${body}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SOURCE}")
add_executable(apu_idle_test apu_idle_test.c)
target_include_directories(apu_idle_test PRIVATE "${APU}" "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME apu_idle COMMAND apu_idle_test)
