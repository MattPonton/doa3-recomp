cmake_minimum_required(VERSION 3.20)
project(point_sprite_test C)
enable_testing()
set(NV "${CMAKE_CURRENT_SOURCE_DIR}/../../src/nv2a/nv2a_pgraph_d3d11.c")
set(D3D "${CMAKE_CURRENT_SOURCE_DIR}/../../src/d3d")
file(READ "${NV}" content)
function(extract first last output)
  string(FIND "${content}" "${first}" begin)
  string(FIND "${content}" "${last}" end)
  if(begin LESS 0 OR end LESS begin)
    message(FATAL_ERROR "Point-sprite extraction boundary changed")
  endif()
  math(EXPR length "${end}-${begin}")
  string(SUBSTRING "${content}" ${begin} ${length} part)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}" "${part}")
endfunction()
extract("/* XYZRHW/diffuse/UV0 prefix" "static struct {" vertex.h)
extract("static uint32_t nv_xbox_texop(" "/* Draw from the bound vertex arrays" texop.h)
extract("static int nv_apply_point_state(" "/* Inline (DrawVerticesUP)" points.h)
extract("static void nv_comb_input(" "/* Specular power from SET_SPECULAR_PARAMS" decode.h)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${NV}")
add_executable(point_sprite_test point_sprite_test.c "${D3D}/d3d8_combiners.c")
target_include_directories(point_sprite_test PRIVATE "${D3D}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(point_sprite_test PRIVATE d3d11 d3dcompiler dxguid)
add_test(NAME point_sprite_warp COMMAND point_sprite_test)
