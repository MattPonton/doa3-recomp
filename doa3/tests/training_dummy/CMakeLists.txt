cmake_minimum_required(VERSION 3.20)
project(training_dummy_test C)
enable_testing()
# Exercise the actual native input converters without linking the whole game.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/game/recomp/gen/recomp_0000.c" lifted)
set(native "#define RECOMP_GENERATED_CODE\n#include \"recomp/recomp_types.h\"\n")
foreach(symbol IN ITEMS sub_00011000_gen sub_0001EF20_gen)
    string(FIND "${lifted}" "void ${symbol}(void)" start)
    if(start LESS 0)
        message(FATAL_ERROR "Missing 3.1 native input converter: ${symbol}")
    endif()
    string(SUBSTRING "${lifted}" ${start} -1 tail)
    string(FIND "${tail}" "/**" finish)
    string(SUBSTRING "${tail}" 0 ${finish} body)
    string(APPEND native "${body}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/native_input.c" "${native}")
add_executable(training_dummy_test training_dummy_test.c ../../src/game/training_dummy.c "${CMAKE_CURRENT_BINARY_DIR}/native_input.c")
target_include_directories(training_dummy_test PRIVATE ../../src/game ../../src)
add_test(NAME training_dummy COMMAND training_dummy_test)
