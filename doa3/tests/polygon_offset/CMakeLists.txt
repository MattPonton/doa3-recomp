cmake_minimum_required(VERSION 3.20)
project(polygon_offset_test C)
enable_testing()
function(extract file first last output)
 file(READ "${file}" content)
 string(FIND "${content}" "${first}" begin)
 string(FIND "${content}" "${last}" end)
 if(begin LESS 0 OR end LESS begin)
  message(FATAL_ERROR "Polygon offset extraction boundary changed: ${file}")
 endif()
 math(EXPR length "${end}-${begin}")
 string(SUBSTRING "${content}" ${begin} ${length} part)
 file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}" "${part}")
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${file}")
endfunction()
set(D3D "${CMAKE_CURRENT_SOURCE_DIR}/../../src/d3d")
set(NV "${CMAKE_CURRENT_SOURCE_DIR}/../../src/nv2a")
extract("${D3D}/d3d8_states.c" "static D3D11_RASTERIZER_DESC g_last_raster_desc;" "/* ================================================================\n * D3D8" offset.h)
extract("${D3D}/d3d8_states.c" "static void update_rasterizer_state(" "/* ================================================================\n * Sampler" raster.h)
extract("${NV}/nv2a_pgraph_d3d11.c" "static void nv_apply_polygon_offset(" "static void nv_apply_draw_state(" pgraph.h)
add_executable(polygon_offset_test polygon_offset_test.c)
target_include_directories(polygon_offset_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${NV}")
target_link_libraries(polygon_offset_test PRIVATE d3d11 d3dcompiler)
add_test(NAME polygon_offset COMMAND polygon_offset_test)
