cmake_minimum_required(VERSION 3.20)
project(nv2aff_draw_test C)
enable_testing()
set(D3D "${CMAKE_CURRENT_SOURCE_DIR}/../../src/d3d")
file(READ "${D3D}/d3d8_device.c" SOURCE)
string(FIND "${SOURCE}" "static int nv2aff_ring_upload(" BEGIN)
if(BEGIN LESS 0)
    message(FATAL_ERROR "NV2A draw extraction boundary changed")
endif()
string(SUBSTRING "${SOURCE}" ${BEGIN} -1 DRAW)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/draw_impl.h" "${DRAW}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${D3D}/d3d8_device.c")
add_executable(nv2aff_draw_test nv2aff_draw_test.c "${D3D}/d3d8_shaders.c"
    "${D3D}/d3d8_combiners.c" "${D3D}/d3d8_shader_cache.c")
target_include_directories(nv2aff_draw_test PRIVATE "${D3D}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(nv2aff_draw_test PRIVATE d3d11 d3dcompiler dxguid bcrypt)
add_test(NAME nv2aff_draw_warp COMMAND nv2aff_draw_test)

target_sources(nv2aff_draw_test PRIVATE ../../src/perf_capture.c)
