cmake_minimum_required(VERSION 3.20)
project(texture_cache_test C)
enable_testing()
set(NV "${CMAKE_CURRENT_SOURCE_DIR}/../../src/nv2a/nv2a_pgraph_d3d11.c")
file(READ "${NV}" SOURCE)
string(REGEX MATCH "#define TEXCACHE_N[^\n]*\n#define TEXCACHE_HASH_N[^\n]*\n#define TEXCACHE_BUDGET[^\n]*" LIMITS "${SOURCE}")
string(FIND "${SOURCE}" "static uint32_t texcache_hash(" BEGIN)
string(FIND "${SOURCE}" "static IDirect3DTexture8 *get_dynamic_texture(" END)
if(BEGIN LESS 0 OR END LESS BEGIN OR NOT LIMITS)
    message(FATAL_ERROR "Texture cache extraction boundary changed")
endif()
math(EXPR LENGTH "${END}-${BEGIN}")
string(SUBSTRING "${SOURCE}" ${BEGIN} ${LENGTH} CACHE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/cache_limits.h" "${LIMITS}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/cache_impl.h" "${CACHE}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${NV}")
add_executable(texture_cache_test texture_cache_test.c)
target_include_directories(texture_cache_test PRIVATE "../../src/d3d" "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME texture_cache COMMAND texture_cache_test)
